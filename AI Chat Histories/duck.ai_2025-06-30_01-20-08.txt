o4-mini:
In GIMP’s PDB everything that a procedure “returns” is actually declared up‐front when you register the procedure with `gimp_install_procedure()` (in C) or `register()` (in Python‐fu), not something you can go and mutate later via `gimp_procedure_find_return_value()`. That function simply lets you *read* the metadata of an already‐registered procedure, not change it.

The return‐values of your plugin are defined by the array of `GimpParamDef` you pass — that array is part of your call to `gimp_install_procedure()`. Here’s the typical pattern in C:

```c
#include <libgimp/gimp.h>

static void
query (void)
{
  static GimpParamDef args[] = {
    { GIMP_PDB_IMAGE,     "image",    "Input image"     },
    { GIMP_PDB_DRAWABLE,  "drawable", "Input drawable"  }
  };

  /* status code always first return */
  static GimpParamDef return_vals[] = {
    { GIMP_PDB_STATUS,    "status",   "Return status" },
    { GIMP_PDB_IMAGE,     "out_image","Processed image" }  /* optional extra */
  };

  gimp_install_procedure (
    "plug-in-myfilter",               /* Proc name */
    "My Filter",                      /* Blurb */
    "Does something wonderful",       /* Help */
    "Your Name <you@domain.com>",     /* Author */
    "Your Name",                      /* Copyright holder */
    "2024",                           /* Date */
    "_My Filter…",                    /* Menu label */
    "RGB*, GRAY*",                    /* Image types */
    GIMP_PLUGIN,                      /* Proc type */
    G_N_ELEMENTS (args),              /* # of args */
    G_N_ELEMENTS (return_vals),       /* # of return values */
    args,
    return_vals
  );
}

static void
run (const gchar      *name,
     gint              nparams,
     const GimpParam  *param,
     gint             *nreturn_vals,
     GimpParam       **return_vals)
{
  static GimpParam  values[2];

  *nreturn_vals = 2;
  *return_vals  = values;

  /* always return a status */
  values[0].type = GIMP_PDB_STATUS;
  values[0].data.d_status = GIMP_PDB_SUCCESS;

  /* then any extra return(s) in the order you declared them */
  values[1].type = GIMP_PDB_IMAGE;
  values[1].data.d_image = your_output_image_id;
}

GIMP_PLUGIN_DEFINE (
  /* init_proc  */ NULL,
  /* quit_proc  */ NULL,
  /* query_proc */ query,
  /* run_proc   */ run
)
```

Key takeaways:

  1. The *shape* (number and types) of return values is fixed at `gimp_install_procedure()` time.
  2. At run‐time you fill in those slots in your `run()` function: the first slot is always a status code (`GIMP_PDB_SUCCESS` or an error), and thereafter any additional return values you declared.
  3. You cannot go and “set” or “redefine” return slots via `gimp_procedure_find_return_value()`—that API is read‐only.

If you’re writing Python‐fu it’s exactly the same idea, but you call:

```python
register(
    "python-fu-myfilter",
    "My Filter",
    "Does something wonderful",
    "Your Name",
    "Your Name",
    "2024",
    "<Image>/Filters/My Filter…",
    "RGB*, GRAY*",
    [
      (PF_IMAGE,    "image",    "Input image",    None),
      (PF_DRAWABLE, "drawable", "Input drawable", None),
    ],
    [
      (PF_IMAGE,    "out_image","Output image"),
    ],
    myfilter_func
)
main()
```

Here the last two lists are your argument descriptors and your return‐value descriptors. The first return you must emit is always the status code; in Python‐fu that’s handled for you under the hood (you simply return a tuple whose first element is the new image or drawable).

```c
#include <libgimp/gimp.h>

static void
query (Gimp::Procedure *proc)
{
  /* tell GIMP what our proc is called, what it takes, and what it returns */
  gimp_procedure_set_buildable (proc, TRUE);  /* interactive */
  gimp_procedure_set_menu_label (proc, "_My Filter…");
  gimp_procedure_set_image_types (proc, "RGB*, GRAY*");

  /* arguments */
  gimp_procedure_set_n_params (proc, 2);
  gimp_procedure_set_param_info (proc, 0,
                                 GIMP_PARAM_TYPE_IMAGE,
                                 "image",
                                 "Input image");
  gimp_procedure_set_param_info (proc, 1,
                                 GIMP_PARAM_TYPE_DRAWABLE,
                                 "drawable",
                                 "Input drawable");

  /* return values */
  gimp_procedure_set_n_return_vals (proc, 2);
  gimp_procedure_set_return_val_info (proc, 0,
                                      GIMP_PARAM_TYPE_STATUS,
                                      "status",
                                      "Success or Failure");
  gimp_procedure_set_return_val_info (proc, 1,
                                      GIMP_PARAM_TYPE_IMAGE,
                                      "new-image",
                                      "The newly created image");

  gimp_procedure_done (proc);
}

static GimpRunReturn
run (Gimp::Procedure   *proc,
     GimpRunContext    *run_context,
     unsigned           n_params,
     const GimpParam   *param,
     unsigned          *n_return_vals,
     GimpParam        **return_vals)
{
  GimpParam *retvals;
  GimpImage *in_image    = gimp_image_get (param[0].data.d_image);
  GimpDrawable *in_draw  = gimp_drawable_get (param[1].data.d_drawable);

  /* --- 1) do your work here, create a new image --- */
  GimpImage *out_image = gimp_image_new (gimp_image_get_width (in_image),
                                         gimp_image_get_height (in_image),
                                         gimp_image_get_base_type (in_image));
  /* ... paint into out_image, add layers, whatever ... */

  /* --- 2) allocate & fill return-value array --- */
  *n_return_vals = 2;
  retvals = g_new0 (GimpParam, *n_return_vals);

  /* always return a status code first */
  retvals[0].type           = GIMP_PARAM_TYPE_STATUS;
  retvals[0].data.d_status  = GIMP_RUN_SUCCESS;

  /* then any extra return values in the order you declared them */
  retvals[1].type           = GIMP_PARAM_TYPE_IMAGE;
  /* here you must return the *ID* of the newly created image: */
  retvals[1].data.d_image   = gimp_image_get_id (out_image);

  *return_vals = retvals;

  return GIMP_RUN_SUCCESS;
}

GIMP_PLUGIN_DEFINE (
  /* name        */ "plug-in-myfilter",
  /* blurb       */ "My awesome filter",
  /* help        */ "Help text for My Filter",
  /* author      */ "Me <me@example.com>",
  /* copyright   */ "Me 2024",
  /* date        */ "2024",
  /* proc_type   */ GIMP_PLUGIN,
  /* image_types */ "",
  /* query_proc  */ query,
  /* run_proc    */ run
)
```

Key points:

 1.  In `query()` you call `gimp_procedure_set_n_return_vals()` and `gimp_procedure_set_return_val_info()` to declare that you will return two parameters:  
    • slot 0: a `GIMP_PARAM_TYPE_STATUS`.  
    • slot 1: a `GIMP_PARAM_TYPE_IMAGE`.  
 2.  In `run()` you allocate an array of `GimpParam` of exactly that size, set the first element to `GIMP_RUN_SUCCESS` (or an error), and set the second to the *ID* of the `GimpImage` you created.  
 3.  Assign your array into `*return_vals` and set `*n_return_vals` to the array length.  

That’s it.  Once you do that, scripting callers (and non-interactive PDB callers) will see your new image returned as the second return parameter.

```c
GIMP_PLUGIN_ENTRY_POINT
{
  plugin_init ();

                    /* argv = */ NULL,

}

/* 1) INFERENCE / REGISTRATION */

plugin_init (void)
{
  /* Nothing special here for 3.x: GimpInitializer is handled for you. */
}

/* 2) CREATE AND DESCRIBE YOUR PROCEDURE */
static GimpProcedure *
plugin_create_procedure (void)
{
  GimpProcedure *proc;

  proc = gimp_procedure_new ("plug-in-myfilter3",
                             "My Filter 3.x",

                             "Me <me@example.com>",

  gimp_procedure_add_image_return_value (proc, "new_image",

  gimp_procedure_add_menu_path (proc,   "<Image>/Filters/MyFilter3…");

plugin_run (GimpProcedure    *proc,
            GimpRunMode       run_mode,
            GimpImage        *in_image,
            GimpDrawable    **in_drawables,
```
